cmake_minimum_required(VERSION 3.24)
project(TorchBridgeNativeInput LANGUAGES C CXX)
enable_testing()
add_executable(input_policy_test policy_test.cpp)
target_compile_features(input_policy_test PRIVATE cxx_std_17)
add_test(NAME input_policy COMMAND input_policy_test)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(STATUS "Only the portable input policy test is built on this platform")
    return()
endif()
add_library(tb_minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
    vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde32.c)
target_include_directories(tb_minhook PUBLIC vendor/minhook/include)
add_library(torchbridge_input SHARED proxy.cpp runtime.cpp linux_memory.cpp winmm.def)
add_executable(input_loader_test proxy_test.cpp)
target_compile_features(input_loader_test PRIVATE cxx_std_17)
add_test(NAME input_loader COMMAND input_loader_test $<TARGET_FILE:torchbridge_input>)
add_executable(input_interaction_test interaction_test.cpp)
target_compile_features(input_interaction_test PRIVATE cxx_std_17)
target_compile_definitions(input_interaction_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(input_interaction_test PRIVATE tb_minhook user32 ws2_32 bcrypt)
add_test(NAME input_interaction COMMAND input_interaction_test)
target_compile_features(torchbridge_input PRIVATE cxx_std_17)
target_compile_definitions(torchbridge_input PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(torchbridge_input PRIVATE tb_minhook user32 ws2_32 bcrypt)
set_target_properties(torchbridge_input PROPERTIES PREFIX "" OUTPUT_NAME "winmm")
foreach(target torchbridge_input tb_minhook input_policy_test input_loader_test input_interaction_test)
    set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
if(MSVC)
    target_compile_options(torchbridge_input PRIVATE /W4 /EHsc)
else()
    target_compile_options(torchbridge_input PRIVATE -Wall -Wextra -Werror)
    target_link_options(torchbridge_input PRIVATE -static -static-libgcc -static-libstdc++)
endif()
