cmake_minimum_required(VERSION 3.24)
project(TorchBridgeFullscreen LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Torchlight 1 requires a Windows x86 build (Visual Studio: -A Win32)")
endif()
enable_testing()
add_executable(protocol_test protocol_test.cpp)
target_compile_features(protocol_test PRIVATE cxx_std_17)
add_test(NAME wire_protocol COMMAND protocol_test)
add_executable(proxy_smoke_test proxy_smoke_test.cpp)
target_compile_features(proxy_smoke_test PRIVATE cxx_std_17)
target_link_libraries(proxy_smoke_test PRIVATE user32)
set_target_properties(proxy_smoke_test PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MINGW)
    target_link_options(proxy_smoke_test PRIVATE -static)
endif()
add_test(NAME d3d9_proxy COMMAND proxy_smoke_test)
add_library(torchbridge_d3d9 SHARED proxy.cpp renderer.cpp d3d9.def)
target_compile_features(torchbridge_d3d9 PRIVATE cxx_std_17)
target_compile_definitions(torchbridge_d3d9 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(torchbridge_d3d9 PRIVATE user32)
set_target_properties(torchbridge_d3d9 PROPERTIES PREFIX "" OUTPUT_NAME "d3d9"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    target_compile_options(torchbridge_d3d9 PRIVATE /W4 /EHsc)
else()
    target_compile_options(torchbridge_d3d9 PRIVATE -Wall -Wextra -Werror)
    target_link_options(torchbridge_d3d9 PRIVATE -static -static-libgcc -static-libstdc++)
endif()
